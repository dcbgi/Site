rules_version = '2';
// Firestore security rules for the Book Challenge experience.
// Paste this into: Firebase Console → Firestore Database → Rules → Publish.
//
// Model: fully public read (everyone can see the scoreboard and entries),
// public create (anyone can log a book for any reader, or add a reader),
// but delete is restricted to the anonymous session that created the
// document — so only "you" (this browser/device) can delete what you added.
service cloud.firestore {
  match /databases/{database}/documents {

    match /readers/{readerId} {
      allow read: if true;
      allow create: if request.auth != null
                    && request.resource.data.creatorUid == request.auth.uid
                    && request.resource.data.name is string
                    && request.resource.data.name.size() > 0
                    && request.resource.data.goalBooks is int
                    && request.resource.data.goalBooks > 0;
      // Anyone signed in (anonymously) may update a reader's goal.
      allow update: if request.auth != null
                    && request.resource.data.goalBooks is int
                    && request.resource.data.goalBooks > 0;
      // Removing a reader could orphan their entries — manage that from
      // the Firebase console instead of exposing it in the UI.
      allow delete: if false;
    }

    match /entries/{entryId} {
      allow read: if true;
      allow create: if request.auth != null
                    && request.resource.data.creatorUid == request.auth.uid
                    && request.resource.data.readerName is string
                    && request.resource.data.title is string
                    && request.resource.data.title.size() > 0
                    && request.resource.data.pages is int
                    && request.resource.data.pages > 0;
      allow update: if false;
      allow delete: if request.auth != null
                    && resource.data.creatorUid == request.auth.uid;
    }
  }
}
