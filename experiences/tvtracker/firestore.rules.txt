rules_version = '2';
// Firestore security rules for the TV Show Tracker experience.
// If reusing the Book Challenge project, add this `match /shows/{showId}`
// block inside the existing `service cloud.firestore { match /databases/... }`
// alongside the readers/entries rules, then Publish.
//
// Model: fully public read, public create/update (anyone can add or update
// progress on any show, for any username), but delete is restricted to the
// anonymous session that created the document.
service cloud.firestore {
  match /databases/{database}/documents {

    match /shows/{showId} {
      allow read: if true;

      allow create: if request.auth != null
                    && request.resource.data.creatorUid == request.auth.uid
                    && request.resource.data.username is string
                    && request.resource.data.username.size() > 0
                    && request.resource.data.title is string
                    && request.resource.data.title.size() > 0
                    && request.resource.data.status in ['want_to_watch', 'watching', 'completed']
                    && request.resource.data.season is int && request.resource.data.season > 0
                    && request.resource.data.episode is int && request.resource.data.episode > 0
                    && request.resource.data.rating is int
                    && request.resource.data.rating >= 0 && request.resource.data.rating <= 5;

      // Anyone signed in (anonymously) may update status/progress/rating —
      // this is a shared board, not a per-user private list. creatorUid
      // itself can't be reassigned, so delete permission can't be hijacked.
      allow update: if request.auth != null
                    && request.resource.data.creatorUid == resource.data.creatorUid
                    && request.resource.data.status in ['want_to_watch', 'watching', 'completed']
                    && request.resource.data.season is int && request.resource.data.season > 0
                    && request.resource.data.episode is int && request.resource.data.episode > 0
                    && request.resource.data.rating is int
                    && request.resource.data.rating >= 0 && request.resource.data.rating <= 5;

      allow delete: if request.auth != null
                    && resource.data.creatorUid == request.auth.uid;
    }
  }
}
